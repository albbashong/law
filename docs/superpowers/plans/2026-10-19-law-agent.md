# 법령 근거 에이전트(law_review) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 인허가 질의에 대해 법령·조례·별표를 스스로 따라가며 근거 조문을 인용해 답하는 A2A 에이전트(law_review)와 그 도구 서버(law-mcp)를 만든다.

**Architecture:** 오프라인 스크립트가 수집 원본(XML·별표 텍스트·연계 데이터)을 `data/lawdb.sqlite` 하나로 색인한다. law-mcp(공식 mcp SDK v2, streamable HTTP)가 그 DB를 읽기 전용으로 열어 도구 3개와 따라가기 엔진을 제공한다. law_review는 permit-mcp-agent의 공용 A2A 런타임 복사본 위에 agent.yaml/prompt.txt/.env만 얹어 gpt-oss-120b로 도구를 부른다.

**Tech Stack:** Python 3.12, uv, sqlite3(FTS5 trigram, SQLite 3.50), mcp>=2.0.0,<3, uvicorn, pydantic-settings, pyyaml, pytest. 런타임: fastapi, fastmcp>=3.3.1,<4, openai.

**Spec:** `docs/superpowers/specs/2026-10-19-law-agent-design.md`

## Global Constraints

- 결과 형식: `{status, facts, message, steps}`; status ∈ `ok | partial | needs_disambiguation | failed`.
- `facts` 직렬화 길이 ≤ 7,600자 (`FACTS_MAX_CHARS=7600`).
- 도구 설명 docstring 첫 문단(앞 160자)에 사용자 표현: "지을 수 있어", "건폐율", "근거 법령", "조례".
- `refs[]` 항목 형식: `{law, article, title, excerpt, note, doc_id, path}`.
- 근거상태 값: `시행령_허용 | 조례_허용 | 조례_미포함 | 미열거`.
- 법령 데이터는 실행 중 외부 API를 호출하지 않는다(DB 읽기 전용 `mode=ro`). 외부 호출은 `parcel_zoning`의 브이월드뿐.
- 독립 동작: 주소만 담긴 질의에 에이전트 혼자 답할 수 있어야 한다.
- 포트: law-mcp 7017, law_review 7016.
- 키는 `.env`(LAW_API_OC)에서 읽는다. 코드·문서에 키를 쓰지 않는다.
- 가능/불가를 단정하는 문구를 도구 message·에이전트 답에 쓰지 않는다.
- 런타임 복사본(`agent/runtime`)은 출처 커밋 `permit-mcp-agent cac8607`을 README에 기록하고 수정 최소화.

```python
import re
import xml.etree.ElementTree as ET
from dataclasses import dataclass

CIRCLED = {c: i for i, c in enumerate("①②③④⑤⑥⑦⑧⑨⑩⑪⑫⑬⑭⑮⑯⑰⑱⑲⑳", 1)}
HO_RE = re.compile(r"^\s*(\d+)(?:의(\d+))?\.\s")
MOK_RE = re.compile(r"^\s*([가-하])\.\s")
HANG_RE = re.compile(r"^\s*([①-⑳])")

@dataclass
class Unit:
    path: str
    label: str
    title: str
    text: str
    parent: str | None

def _jo_key(no: str, branch: str | None) -> str:
    no = str(int(no))
    return f"제{no}조" + (f"의{int(branch)}" if branch and int(branch) else "")

def _ho_no(text: str) -> str | None:
    m = HO_RE.match(text)
    return (m.group(1) + (f"의{m.group(2)}" if m.group(2) else "")) if m else None

def law_units(root: ET.Element) -> list[Unit]:
    out: list[Unit] = []
    for ju in root.iter("조문단위"):
        if ju.findtext("조문여부") != "조문":
            continue
        jo = _jo_key(ju.findtext("조문번호"), ju.findtext("조문가지번호"))
        title = (ju.findtext("조문제목") or "").strip()
        body = [(ju.findtext("조문내용") or "").strip()]
        hangs = ju.findall("항")
        for hi, h in enumerate(hangs, 1):
            hno = CIRCLED.get((h.findtext("항번호") or "").strip()[:1], hi if len(hangs) > 1 else None)
            hseg = str(hno) if hno else "-"
            htext = (h.findtext("항내용") or "").strip()
            hpath = f"{jo}/{hseg}"
            if htext:
                body.append(htext)
                out.append(Unit(hpath, f"{jo}제{hno}항" if hno else jo, title, htext, jo))
            for ho in h.findall("호"):
                t = (ho.findtext("호내용") or "").strip()
                n = _ho_no(t) or (ho.findtext("호번호") or "").strip(". ")
                ppath = f"{hpath}/{n}"
                out.append(Unit(ppath, f"{jo}{f'제{hno}항' if hno else ''}제{n}호", title, t, hpath))
                body.append(t)
                for mk in ho.findall("목"):
                    mt = (mk.findtext("목내용") or "").strip()
                    m = MOK_RE.match(mt)
                    mn = m.group(1) if m else (mk.findtext("목번호") or "").strip(". ")
                    out.append(Unit(f"{ppath}/{mn}", f"{jo}{f'제{hno}항' if hno else ''}제{n}호{mn}목",
                                    title, mt, ppath))
                    body.append(mt)
        out.insert(len(out) - sum(1 for u in out if u.path.startswith(jo + "/")),
                   Unit(jo, jo, title, "\n".join(b for b in body if b), None))
    return out

def ordin_units(root: ET.Element) -> list[Unit]:
    out: list[Unit] = []
    for z in root.iter("조"):
        if z.findtext("조문여부") not in ("Y", "조문"):
            continue
        raw = z.findtext("조문번호") or ""
        if not raw.isdigit() or int(raw[:4]) == 0:
            continue
        jo = _jo_key(raw[:4], raw[4:6])
        title = (z.findtext("조제목") or "").strip()
        text = (z.findtext("조내용") or "").strip()
        out.append(Unit(jo, jo, title, text, None))
        hang, ho = None, None
        for line in text.split("\n"):
            line = line.strip()
            if not line:
                continue
            hm = HANG_RE.match(line)
            if hm:
                hang, ho = CIRCLED[hm.group(1)], None
                out.append(Unit(f"{jo}/{hang}", f"{jo}제{hang}항", title, line, jo))
                continue
            n = _ho_no(line)
            hseg = str(hang) if hang else "-"
            if n:
                ho = n
                out.append(Unit(f"{jo}/{hseg}/{n}", f"{jo}{f'제{hang}항' if hang else ''}제{n}호",
                                title, line, f"{jo}/{hang}" if hang else jo))
                continue
            mm = MOK_RE.match(line)
            if mm and ho:
                out.append(Unit(f"{jo}/{hseg}/{ho}/{mm.group(1)}",
                                f"{jo}{f'제{hang}항' if hang else ''}제{ho}호{mm.group(1)}목",
                                title, line, f"{jo}/{hseg}/{ho}"))
    return out
```

```python
import re

LEVELS = [
    ("ho", re.compile(r"^\s*(\d+)(?:의(\d+))?\.\s"), lambda m: m.group(1) + (f"의{m.group(2)}" if m.group(2) else ""), "제{}호"),
    ("mok", re.compile(r"^\s*([가-하])\.\s"), lambda m: m.group(1), "{}목"),
    ("p1", re.compile(r"^\s*\((\d+)\)\s?"), lambda m: m.group(1), "({})"),
    ("p2", re.compile(r"^\s*\(([가-하])\)\s?"), lambda m: m.group(1), "({})"),
]

def _norm(text: str) -> list[str]:
    # 표 셀 줄바꿈으로 끊긴 문장을 이어 붙인다: 번호로 시작하지 않는 줄은 앞 줄에 붙인다
    lines, buf = [], ""
    for raw in text.splitlines():
        s = re.sub(r"\s+", " ", raw).strip()
        if not s:
            continue
        if buf and not any(rx.match(s) for _, rx, _, _ in LEVELS):
            buf += " " + s
        else:
            if buf:
                lines.append(buf)
            buf = s
    if buf:
        lines.append(buf)
    return lines

def split_annex(text: str) -> list[tuple[str, str, str]]:
    stack: list[tuple[int, str, str]] = []   # (level, seg, labelpart)
    items: dict[str, list] = {}
    order: list[str] = []
    for line in _norm(text):
        hit = None
        for lv, (_, rx, key, fmt) in enumerate(LEVELS):
            m = rx.match(line)
            if m:
                hit = (lv, key(m), fmt.format(key(m)))
                break
        if hit:
            lv, seg, lab = hit
            while stack and stack[-1][0] >= lv:
                stack.pop()
            stack.append(hit)
            path = "/".join(s for _, s, _ in stack)
            items[path] = [" ".join(l for _, _, l in stack), [line]]
            order.append(path)
        for i in range(1, len(stack) + 1 if not hit else len(stack)):
            p = "/".join(s for _, s, _ in stack[:i])
            if p in items:
                items[p][1].append(line)
    return [(p, items[p][0], "\n".join(items[p][1])) for p in order]
```

```python
import re
from dataclasses import dataclass

LAW = r"「(?P<law>[^」]+)」\s*"
SAME = r"(?P<same>같은\s*법(?:\s*(?P<same_sfx>시행령|시행규칙))?)\s*"
REL = r"(?P<rel>법|영|규칙|시행령|시행규칙)\s*"
JO = r"제\s*(?P<jo>\d+)\s*조(?:의\s*(?P<jo2>\d+))?(?:\s*제\s*(?P<hang>\d+)\s*항)?(?:\s*제\s*(?P<ho>\d+)\s*호)?(?:\s*(?P<mok>[가-하])\s*목)?"
ANNEX = r"별표\s*(?P<an>\d+)(?:의\s*(?P<an2>\d+))?(?:\s*제\s*(?P<aho>\d+)\s*호)?(?:\s*(?P<amok>[가-하])\s*목)?"
RX = re.compile(rf"(?:{LAW}|{SAME}|{REL})?(?:{JO}|{ANNEX})")

@dataclass
class Citation:
    law: str | None; rel: str | None; jo: str | None; hang: str | None; ho: str | None
    mok: str | None; annex: str | None; annex_item: str | None; span: str

def find_citations(text: str) -> list[Citation]:
    out, last_law = [], None
    for m in RX.finditer(text):
        law, rel = m.group("law"), m.group("rel")
        if law:
            last_law = law.strip()
        elif m.group("same") and last_law:
            base = re.sub(r"\s*시행(령|규칙)$", "", last_law)
            law = base + (f" {m.group('same_sfx')}" if m.group("same_sfx") else "")
        if rel in ("시행령",):
            rel = "영"
        if rel == "시행규칙":
            rel = "규칙"
        if not (law or rel):
            continue   # 맥락 없는 "제3조"는 같은 문서 내부 참조 — Task 5에서 self 링크로 처리
        jo = None
        if m.group("jo"):
            jo = f"제{m.group('jo')}조" + (f"의{m.group('jo2')}" if m.group("jo2") else "")
        annex = None
        if m.group("an"):
            annex = m.group("an") + (f"의{m.group('an2')}" if m.group("an2") else "")
        item = "/".join(x for x in (m.group("aho"), m.group("amok")) if x) or None
        out.append(Citation(law and law.strip(), rel, jo, m.group("hang"), m.group("ho"),
                            m.group("mok"), annex, item, m.group(0)))
    return out

def citation_path(c: Citation) -> str:
    if c.annex:
        return f"별표{c.annex}" + (f":{c.annex_item}" if c.annex_item else "")
    segs = [c.jo]
    if c.hang or c.ho:
        segs.append(c.hang or "-")
    if c.ho:
        segs.append(c.ho)
    if c.mok:
        segs.append(c.mok)
    return "/".join(segs)
```
